cmake_minimum_required(VERSION 3.22)
project(SpoolCutQtPrototype VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets OpenGLWidgets)
find_package(PkgConfig REQUIRED)
if(WIN32)
    # MSYS2's mpv.pc reports /ucrt64 paths. Resolve the header and import
    # library from the native compiler prefix so CMake gets Windows paths.
    pkg_check_modules(MPV REQUIRED mpv>=2.3)
    get_filename_component(ucrt_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    get_filename_component(ucrt_prefix "${ucrt_bin}" DIRECTORY)
    find_path(MPV_HEADER_DIR mpv/client.h HINTS "${ucrt_prefix}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(MPV_IMPORT_LIBRARY mpv HINTS "${ucrt_prefix}/lib" NO_DEFAULT_PATH REQUIRED)
    add_library(mpv-native UNKNOWN IMPORTED)
    set_target_properties(mpv-native PROPERTIES IMPORTED_LOCATION "${MPV_IMPORT_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${MPV_HEADER_DIR}")
    set(mpv_target mpv-native)
else()
    pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv>=2.3)
    set(mpv_target PkgConfig::MPV)
endif()

add_subdirectory(../cpp/core shared-core)

add_library(prototype-widgets STATIC editor.cpp preview.cpp timeline.cpp window.cpp)
target_link_libraries(prototype-widgets PUBLIC spoolcut-core Qt6::Widgets Qt6::OpenGLWidgets ${mpv_target})
spoolcut_target(prototype-widgets)
target_include_directories(prototype-widgets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(prototype-widgets PRIVATE appimage_launcher.cpp)
endif()

qt_add_executable(spoolcut-qt WIN32 main.cpp)
qt_add_resources(spoolcut-qt "spoolcut-icon"
    PREFIX "/" BASE "../assets" FILES "../assets/io.github.bugrakoc.SpoolCut.png")
spoolcut_target(spoolcut-qt)
target_link_libraries(spoolcut-qt PRIVATE prototype-widgets)
if(WIN32)
    enable_language(RC)
    target_sources(spoolcut-qt PRIVATE spoolcut.rc)
    target_include_directories(spoolcut-qt PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
    qt_add_executable(prototype-tests tests/prototype_test.cpp)
    spoolcut_target(prototype-tests)
    # Qt Test's timeout macros convert their chrono durations to int.
    target_compile_options(prototype-tests PRIVATE -Wno-conversion)
    target_link_libraries(prototype-tests PRIVATE prototype-widgets Qt6::Test)
    set_target_properties(prototype-tests PROPERTIES WIN32_EXECUTABLE OFF)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        qt_add_resources(prototype-tests "prototype-test-icon"
            PREFIX "/" BASE "../assets" FILES "../assets/io.github.bugrakoc.SpoolCut.png")
    endif()
    set(renderer_modes software auto)
    if(WIN32)
        list(APPEND renderer_modes opengl)
    endif()
    foreach(mode IN LISTS renderer_modes)
        add_test(NAME prototype-${mode} COMMAND prototype-tests)
        set_tests_properties(prototype-${mode} PROPERTIES
            ENVIRONMENT "SPOOLCUT_QT_PREVIEW=${mode};SPOOLCUT_QT_TEST_AUDIO_NULL=1"
            TIMEOUT 100)
    endforeach()
endif()
