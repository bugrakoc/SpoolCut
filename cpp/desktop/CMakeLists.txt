set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets OpenGLWidgets)
find_package(PkgConfig REQUIRED)
if(WIN32)
    # MSYS2's mpv.pc reports /ucrt64 paths. Resolve the header and import
    # library from the native compiler prefix so CMake gets Windows paths.
    pkg_check_modules(MPV REQUIRED mpv>=2.3)
    get_filename_component(ucrt_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
    get_filename_component(ucrt_prefix "${ucrt_bin}" DIRECTORY)
    find_path(MPV_HEADER_DIR mpv/client.h HINTS "${ucrt_prefix}/include" NO_DEFAULT_PATH REQUIRED)
    find_library(MPV_IMPORT_LIBRARY mpv HINTS "${ucrt_prefix}/lib" NO_DEFAULT_PATH REQUIRED)
    add_library(mpv-native UNKNOWN IMPORTED)
    set_target_properties(mpv-native PROPERTIES IMPORTED_LOCATION "${MPV_IMPORT_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${MPV_HEADER_DIR}")
    set(mpv_target mpv-native)
else()
    pkg_check_modules(MPV REQUIRED IMPORTED_TARGET mpv>=2.3)
    set(mpv_target PkgConfig::MPV)
endif()

add_library(spoolcut-widgets STATIC editor.cpp preview.cpp timeline.cpp window.cpp)
target_link_libraries(spoolcut-widgets PUBLIC spoolcut-core Qt6::Widgets Qt6::OpenGLWidgets ${mpv_target})
spoolcut_target(spoolcut-widgets)
target_include_directories(spoolcut-widgets PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(spoolcut-widgets PRIVATE appimage_launcher.cpp)
endif()

set(asset_dir "${PROJECT_SOURCE_DIR}/assets")
set(icon "${asset_dir}/io.github.bugrakoc.SpoolCut.png")
qt_add_executable(spoolcut-desktop WIN32 main.cpp)
qt_add_resources(spoolcut-desktop "spoolcut-icon" PREFIX "/" BASE "${asset_dir}" FILES "${icon}")
spoolcut_target(spoolcut-desktop)
target_compile_definitions(spoolcut-desktop PRIVATE SPOOLCUT_VERSION="${SPOOLCUT_VERSION}")
target_link_libraries(spoolcut-desktop PRIVATE spoolcut-widgets)
set_target_properties(spoolcut-desktop PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
if(WIN32)
    spoolcut_windows_resources(spoolcut-desktop "SpoolCut" "io.github.bugrakoc.SpoolCut" "spoolcut.ico")
endif()

if(BUILD_TESTING)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
    qt_add_executable(spoolcut-desktop-tests tests/desktop_test.cpp)
    spoolcut_target(spoolcut-desktop-tests)
    # Qt Test's timeout macros convert their chrono durations to int.
    target_compile_options(spoolcut-desktop-tests PRIVATE -Wno-conversion)
    target_link_libraries(spoolcut-desktop-tests PRIVATE spoolcut-widgets Qt6::Test)
    set_target_properties(spoolcut-desktop-tests PROPERTIES WIN32_EXECUTABLE OFF)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        qt_add_resources(spoolcut-desktop-tests "spoolcut-test-icon" PREFIX "/" BASE "${asset_dir}" FILES "${icon}")
    endif()
    set(renderer_modes software auto)
    if(WIN32)
        list(APPEND renderer_modes opengl)
    endif()
    foreach(mode IN LISTS renderer_modes)
        add_test(NAME desktop-${mode} COMMAND spoolcut-desktop-tests)
        set_tests_properties(desktop-${mode} PROPERTIES
            ENVIRONMENT "SPOOLCUT_QT_PREVIEW=${mode};SPOOLCUT_QT_TEST_AUDIO_NULL=1"
            TIMEOUT 100)
    endforeach()
endif()
